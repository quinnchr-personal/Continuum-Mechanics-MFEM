\section{Description}

\section{Thermo-chemical Decomposition Model}

The pyrolysis model is written in terms of reaction extents
$\chi_i(t)$, $i=1,\dots,N_r$. For each reaction,
\begin{equation}
\frac{\partial \chi_i}{\partial t}
=
\begin{cases}
0, & \theta < \theta_{\mathrm{thr},i}, \\[0.5em]
A_i \exp\!\left(-\dfrac{E_i}{R\,\theta}\right)\,\theta^{n_i}\,(1-\chi_i)^{m_i},
& \theta \ge \theta_{\mathrm{thr},i},
\end{cases}
\qquad i=1,\dots,N_r,
\end{equation}
subject to
\begin{equation}
0 \le \chi_i \le 1.
\end{equation}

The total pyrolysis source is the sum of the per-reaction contributions,
\begin{equation}
\pi_{\mathrm{tot}} = \sum_{i=1}^{N_r} \pi_i,
\end{equation}
with
\begin{equation}
\pi_i
=
\rho_{I,\sigma(i)}\,\varepsilon_{I,\sigma(i)}\,F_i\,
\frac{\partial \chi_i}{\partial t},
\qquad i=1,\dots,N_r,
\end{equation}
where $\chi_i$ is the reaction extent for reaction $i$, $F_i$ is its mass-loss
fraction, and $\sigma(i)$ is the solid phase to which reaction $i$ is attached.
The variable $\chi_i$ is a dimensionless progress variable, where $\chi_i = 0$
denotes the unreacted state for reaction $i$ and $\chi_i = 1$ denotes a fully
completed reaction.

\section{Conservation of Solid Mass}


\begin{equation}
\frac{\partial \rho_{s}}{\partial t} = - \pi_{\mathrm{tot}}
\end{equation}

\section{Conservation of Gas Mass}

\begin{equation}
\frac{\partial}{\partial t}\left(\phi \rho_g\right)
+ \nabla \cdot
\left(
\phi \rho_g \mathbf{v}_g
\right)
- \pi_{\mathrm{tot}}
= 0
\end{equation}

\section{Conservation of Energy}

\begin{equation}
\frac{\partial }{\partial t} \left( \rho e \right) + \nabla \cdot \left( \mathbf{q} + \mathbf{q}_{\mathrm{conv}} \right) = 0
\end{equation}

\section{Material Model}

Ideal gas law for the pyrolysis gas:
\begin{equation}
\rho_g = \frac{p\,M(\theta,p)}{R\,\theta}
\end{equation}

Darcy flow for the gas velocity:
\begin{equation}
\mathbf{v}_g = -\frac{K}{\mu_g \phi} \nabla p
\end{equation}

Fourier's law for the conductive heat flux:
\begin{equation}
\mathbf{q} = -\kappa \nabla \theta
\end{equation}

Convective heat flux for the gas phase:
\begin{equation}
\mathbf{q}_{\mathrm{conv}} = h_g \phi \rho_g \mathbf{v}_g
\end{equation}

Here $M(\theta,p)$, $h_g(\theta,p)$, and $\mu_g(\theta,p)$ denote the pyrolysis-gas
molecular weight, specific enthalpy, and dynamic viscosity.

Internal energy of the mixture:
\begin{equation}
\rho e = \rho_s e_s + \phi \rho_g e_g
\end{equation}

% \begin{equation}
%     e_{s} = h_{s} = \int_{\theta_{\text{ref}}}^{\theta} c_{ps}(\theta') \,d\theta'
% \end{equation}

Degree of Char:
\begin{equation}
\beta = \frac{\rho_s - \rho_v}{\rho_c - \rho_v}
\end{equation}

Here $\rho_v$ and $\rho_c$ denote the virgin and char solid densities, while
$h_v(\theta,p)$, $h_c(\theta,p)$, $c_{pv}(\theta,p)$, $c_{pc}(\theta,p)$, $\kappa_v(\theta,p)$, and
$\kappa_c(\theta,p)$ denote the corresponding virgin and char specific enthalpies,
specific heat capacities, and thermal conductivities. Likewise, $\phi_v$,
$\phi_c$, $K_v$, and $K_c$ denote the virgin and char gas porosities and
permeabilities.

Internal energy of the solid phase:
\begin{equation}
\rho_{s} e_{s} = (1 - \beta) \rho_{v} h_{v} + \beta \rho_{c} h_{c}
\end{equation}

\begin{equation}
    e_{s} = h_{s} = y_{c} h_{c} + y_{v} h_{v}
\end{equation}

\begin{equation}
    y_{c} = \frac{\beta \rho_{c}}{\rho_{s}}, \qquad y_{v} = \frac{(1 - \beta) \rho_{v}}{\rho_{s}}
\end{equation}

Constitutive closures for transport coefficients:
\begin{equation}
\phi(\beta) = (1 - \beta)\phi_v + \beta \phi_c
\end{equation}

\begin{equation}
K(\beta) = (1 - \beta)K_v + \beta K_c
\end{equation}

\begin{equation}
\kappa(\beta, \theta, p) = y_v \kappa_v(\theta,p) + y_c \kappa_c(\theta,p)
\end{equation}

Internal energy of the gas phase:
\begin{equation}
e_{g} = h_{g} - p/\rho_{g}
\end{equation}

Specific heat capacity of the solid phase:
\begin{equation}
\rho_{s} c_{ps} = \beta \rho_{c} c_{pc} + (1 - \beta) \rho_{v} c_{pv}
\end{equation}

\begin{equation}
\frac{\partial }{\partial t} \left( \rho_{s} e_{s} \right) = \frac{\partial }{\partial t} \left( \rho_{s} h_{s} \right) = \left( \rho_{c} h_{c} - \rho_{v} h_{v} \right) \frac{\partial \beta}{\partial t} + \left( (1 - \beta) \rho_{v} c_{pv} + \beta \rho_{c} c_{pc} \right) \frac{\partial \theta}{\partial t}
\end{equation}

\begin{equation}
    \frac{\partial \beta}{\partial t} = \frac{1}{\rho_{c} - \rho_{v}} \frac{\partial \rho_{s}}{\partial t}
\end{equation}

\begin{equation}
\frac{\partial }{\partial t} \left( \rho_{s} h_{s} \right) = \frac{\left( \rho_{c} h_{c} - \rho_{v} h_{v} \right)}{\rho_{c} - \rho_{v}} \frac{\partial \rho_{s}}{\partial t} + \rho_{s} c_{ps} \frac{\partial \theta}{\partial t}
\end{equation}

\begin{equation}
    \bar{h} = \frac{\rho_{c} h_{c} - \rho_{v} h_{v}}{\rho_{c} - \rho_{v}}
\end{equation}

\begin{equation}
\frac{\partial }{\partial t} \left( \rho_{s} h_{s} \right) = \bar{h} \frac{\partial \rho_{s}}{\partial t} + \rho_{s} c_{ps} \frac{\partial \theta}{\partial t} = - \bar{h} \pi_{\mathrm{tot}} + \rho_{s} c_{ps} \frac{\partial \theta}{\partial t}
\end{equation}

\section{Boundary Conditions}

For the time-dependent Dirichlet variant documented here, both primary
unknowns are prescribed on the top boundary:
\begin{equation}
\theta = \theta_D(t),
\qquad
p = p_D(t)
\qquad \text{on } \Gamma_{\mathrm{top}}.
\end{equation}

In the test-driver interpretation, $\theta_D(t)$ is supplied by an external
temperature schedule and $p_D(t)$ is supplied by the pressure schedule. No
surface energy-balance flux law is imposed on $\Gamma_{\mathrm{top}}$ in this
variant, because the top boundary values are enforced directly as essential
time-dependent Dirichlet data.

Any surface recession in this variant is likewise prescribed kinematically
through the ALE boundary motion, rather than being obtained from a separate
surface recession or mass-balance closure.

On the remaining boundaries, writing the gas mass flux as
\begin{equation}
\boldsymbol{m}_g = \phi \rho_g \mathbf{v}_g,
\end{equation}
the equations retain their natural conditions:
\begin{equation}
\boldsymbol{n}\cdot \boldsymbol{m}_g = 0,
\qquad
\boldsymbol{n}\cdot\left(-\kappa \nabla \theta + h_g \boldsymbol{m}_g\right) = 0
\qquad \text{on } \partial \Omega \setminus \Gamma_{\mathrm{top}}.
\end{equation}

\section{Dirichlet Data}

The time dependence is carried entirely by the prescribed boundary data
\begin{equation}
\theta_D = \theta_D(t),
\qquad
p_D = p_D(t).
\end{equation}

In particular, the strong form in the bulk is unchanged, and the boundary input
enters only through the essential constraints applied at the top boundary.


\section{ALE Definitions}

For any Eulerian field $a(\mathbf{x},t)$, its pullback to the reference domain is
denoted by $\widehat{a}(\widehat{\mathbf{x}},t) = a(\mathcal{A}(\widehat{\mathbf{x}},t),t)$.
When a spatial differential operator appears inside a hat, such as
$\widehat{\nabla \theta}$ or $\widehat{\nabla \cdot \mathbf{w}}$, the meaning is:
first evaluate the derivative in the current configuration, then pull the resulting
field back to the reference domain. These objects are therefore not identified with
reference-domain derivatives until the explicit ALE transformation formulas below
are applied.

\begin{equation}
    \mathcal{A}: \widehat{\Omega} \times [0,t_f] \to \Omega(t), \qquad \mathbf{x} = \mathcal{A}(\widehat{\mathbf{x}}, t)
\end{equation}

\begin{equation}
    \mathbf{w}(\mathbf{x},t) = \left.\frac{\partial \mathcal{A}}{\partial t}\right|_{\widehat{\mathbf{x}}}, \qquad
    \widehat{\mathbf{w}}(\widehat{\mathbf{x}},t) = \mathbf{w}(\mathcal{A}(\widehat{\mathbf{x}},t),t)
\end{equation}

\begin{equation}
    \widehat{\mathbf{G}} = \frac{\partial \mathbf{x}}{\partial \widehat{\mathbf{x}}}, \qquad
    \widehat{J} = \det\left( \widehat{\mathbf{G}} \right), \qquad
    \widehat{\mathbf{F}} = \operatorname{cof}\left( \widehat{\mathbf{G}} \right)
    = \widehat{J}\,\widehat{\mathbf{G}}^{-T}
\end{equation}

\begin{equation}
\left.\frac{\partial \psi}{\partial t}\right|_{\widehat{\mathbf{x}}} = \frac{\partial \psi}{\partial t} + \mathbf{w}\cdot\nabla \psi
\end{equation}

\begin{equation}
    \widehat{\nabla \psi} = \frac{1}{\widehat{J}} \widehat{\mathbf{F}}^{T} \widehat{\nabla} \widehat{\psi}
\end{equation}

\begin{equation}
    \widehat{\nabla \cdot \boldsymbol{\psi}} = \frac{1}{\widehat{J}} \widehat{\nabla} \cdot \left( \widehat{\mathbf{F}} \widehat{\boldsymbol{\psi}} \right)
\end{equation}

\begin{equation}
    \frac{\partial \widehat{J}}{\partial t} = \widehat{J}\, \widehat{\nabla \cdot \mathbf{w}}
\end{equation}

\begin{equation}
    \left. \frac{\partial}{\partial t} \left( \widehat{\theta} \widehat{J} \right) \right|_{\widehat{\mathbf{x}}}
    = \left. \frac{\partial \widehat{\theta}}{\partial t} \right|_{\widehat{\mathbf{x}}} \widehat{J}
    + \widehat{\theta} \left. \frac{\partial \widehat{J}}{\partial t} \right|_{\widehat{\mathbf{x}}}
    = \left. \frac{\partial \widehat{\theta}}{\partial t} \right|_{\widehat{\mathbf{x}}} \widehat{J}
    + \widehat{\theta} \widehat{J} \widehat{\nabla \cdot \mathbf{w}}
\end{equation}
